```

├── css/
│   ├── leviathan_store_home.css

│   └── leviathan_store_market.css
│
├── js/
│   └── leviathan_store_home.js
│
└── assets/
    ├── leviathan_store.png           ← Logo
    ├── console/                      ← Gambar konsol gaming
    ├── handphone/                    ← Gambar handphone (nereid/poseidon/leviathan)
    ├── laptop/                       ← Gambar laptop
    ├── set_pc/                       ← Gambar set PC (termasuk sovereign editions)

```
